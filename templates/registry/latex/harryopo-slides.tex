\section{研究背景}
\begin{frame}{问题动机}
\begin{itemize}
  \item 传统文档工具链：AI 直接生成二进制，排版错误率高达 \alert{30\%}
  \item 中文学术场景三重刚需：方正字体规范、公式保真、模板可复用
  \item 核心命题：如何让 AI \textbf{只产出结构化数据}，排版交给引擎？

\begin{frame}{成本衰减模型}
渲染成本随迭代次数 $i$ 指数衰减，$\alpha$ 为模板复用率：
\[ C_i = C_0 \cdot \alpha^{\,i}, \qquad \alpha \in (0,1) \]
\begin{align}
\frac{\partial C_i}{\partial \alpha} &= C_0 \, i \, \alpha^{\,i-1} < 0 \\
\lim_{i \to \infty} C_i &= 0
\end{align}
\note{推导不必逐行讲，落点是"复用模板边际成本趋零"。}
\end{frame}
