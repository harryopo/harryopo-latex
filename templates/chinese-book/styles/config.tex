% config.tex — 中文版配置与本土化字符串

% ── CJK 默认字体 ──
% 原为 Windows 系统字体（Path=C:/Windows/Fonts/ + GBK 名），换机器即失效。
% 改指向项目内嵌字体（templates/fonts/，与 harryopo-base.sty 同一套），
% 保证 skill 整体拷走后仍能编译。路径相对本文件：styles/ → ../fonts/
%
% ⚠️ 加载顺序很关键：kaoCJKsc.sty 里的 \kao@fontmethodModern 会
% \RequirePackage{unicode-math} 并 \setmonofont{Liberation Mono}，
% 那是**在之后**执行的，会覆盖此处设定。main.tex 的顺序是
%   config.tex → \usepackage{kaoWinCJKsc} → kaoCJKsc.sty
% 所以等宽字体必须放在 kaoWinCJKsc **之后**才生效——见文件末尾。
% ⚠️ FZ*JW 只有单一字重，kaobook 却会请求 b/n（粗）m/it（斜）形状
%   （章节标题用 \fzhei\bfseries、\emph 用斜体）。xeCJK 找不到这些形状时
%   回退 nullfont，于是**所有拉丁字符与连字符**在 PDF 里变成 null 空字符
%   （实测 main.pdf 有 1127 个 \x00）。
%   正解：用 AutoFakeBold/FakeSlant 让 xeCJK 用同一字重"伪造"粗斜，
%   而不是去找不存在的粗体字形。Latin 部分交回 Libertinus（有完整字形）。
\setCJKmainfont[Path=../fonts/, Extension=.TTF,
  AutoFakeBold=2.4, AutoFakeSlant=0.2]{FZSSJW}
\setCJKsansfont[Path=../fonts/, Extension=.TTF,
  AutoFakeBold=2.4, AutoFakeSlant=0.2]{FZHTJW}
\setCJKmonofont[Path=../fonts/, Extension=.ttf]{cour}

% ── 中文本土化字符串 ──
\renewcommand{\contentsname}{目\quad 录}
\renewcommand{\listfigurename}{图\quad 录}
\renewcommand{\listtablename}{表\quad 录}
\renewcommand{\lstlistlistingname}{代码\quad 录}
\renewcommand{\indexname}{索\quad 引}
\renewcommand{\bibname}{主要参考文献}
\renewcommand{\figurename}{图}
\renewcommand{\tablename}{表}
\renewcommand{\partname}{第}
\renewcommand{\chaptername}{}
\renewcommand{\appendixname}{附\quad 录}
% \abstractname 由 biblatex 提供，本文档没引 biblatex → 直接 \renewcommand 会报
% "Command \abstractname undefined"。先兜底再改。
\providecommand{\abstractname}{Abstract}
\renewcommand{\abstractname}{摘\quad 要}

% ── 禁用连字符 ──
\defaultfontfeatures{Ligatures={NoCommon, NoContextual, NoHistoric, NoDiscretionary}}
\righthyphenmin=62
\lefthyphenmin=62
\emergencystretch=1em
\tolerance=400

% ── 页眉页脚设置 ──
\clearscrheadfoot
\automark[section]{chapter}
\addtokomafont{pagehead}{\small\fzhei}
\addtokomafont{pagenumber}{\small\fzhei}

\lehead{\leftmark}
