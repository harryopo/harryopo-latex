% ============================================================
% 字体最终设定（必须在 kaoWinCJKsc / unicode-math 之后）
%
% ⚠️ 为什么要覆盖：kaoCJKsc.sty 的 \kao@fontmethodModern 会执行
%     \setromanfont{Libertinus Serif} / \setmonofont{Liberation Mono}，
%    但本机 TinyTeX **没有安装任何 Libertinus 字体文件**
%    （tlmgr 装的 libertinus / libertinus-otf 只有 .sty，没有 .otf）。
%    fontspec 于是回退到 nullfont，所有拉丁字母、数字、连字符在 PDF 里
%    变成 null 空字符——实测 main.pdf 有 1127 个 \x00，整片"乱码"。
%    对照组：只排中文的正文完全正常，只有走 \setmainfont/\ttfamily 的
%    片段（代码、URL、边注里的命令名）出问题。
%
%    正解：改用项目内嵌字体（与 harryopo-base.sty 同一套），不依赖系统字体。
%      正文拉丁 → XITS（已内嵌，含完整拉丁字形）
%      等宽     → cour（Courier New，已内嵌）
%    本文件由 main.tex 在 \usepackage{kaoWinCJKsc} 之后 \input，
%    是最后一个能安全覆盖 \kao@fontmethodModern 的位置。
% ============================================================
\setmainfont[
    Path       = ../fonts/,
    Extension  = .otf,
    BoldFont   = XITS-Bold,
    ItalicFont = XITS-Italic,
    BoldItalicFont = XITS-BoldItalic
]{XITS-Regular}
\setsansfont[
    Path       = ../fonts/,
    Extension  = .otf,
    BoldFont   = texgyreheros-bold,
    ItalicFont = texgyreheros-italic,
    BoldItalicFont = texgyreheros-bolditalic
]{texgyreheros-regular}
% ⚠️ 等宽字体必须给全四个字重。cour（Courier New）在本机只有单一字重，
%   代码块里的 \textbf / \emph 会让 xeCJK 请求 b/n、m/it 形状 → 报
%   "Font shape TU/cour(2)/b/n undefined" 并回退到 m/n，回退后**CJK 族
%   一起失效**，块内中文改由拉丁主字体 XITS 渲染 → 全部缺字形
%   （实测 XITS-Regular/Italic/Bold 三处合计 691 条 Missing character）。
%   JetBrains Mono 是项目内嵌字体（SIL OFL 1.1，允许嵌入分发），
%   Regular/Italic/Bold/BoldItalic 四字重齐全。
\setmonofont[
    Path       = ../fonts/,
    Extension  = .ttf,
    ItalicFont = JetBrainsMono-Italic,
    BoldFont   = JetBrainsMono-Bold,
    BoldItalicFont = JetBrainsMono-BoldItalic
]{JetBrainsMono-Regular}

% ⚠️ 数学字体必须一并切走，否则整份文档连锁崩。
%   kaoCJKsc.sty:1161 的 \setmathfont{Libertinus Math} 指向**本机没有安装**
%   的字体（tlmgr 装的 libertinus-otf 只有 .sty 没有 .otf），unicode-math
%   找不到就回退，于是 \mathrm/\mathbb 等数学族名解析不出度量 —— 实测
%   142 处 "Missing number, treated as zero" + 142 处 "Illegal unit of
%   measure"，并连锁引发 titlesec 49 处 "No format for this command"，
%   最终在图题处 parameter stack 溢出（TeX capacity exceeded）中断编译。
%   连带症状：中文被主字体 XITS（纯拉丁数学字体）渲染 → 1258 条
%   Missing character，中文整段从 PDF 里丢失。
%   正解：换成项目内嵌的 XITSMath 三件套（含 Math-Bold）。
\setmathfont[
    Path       = ../fonts/,
    Extension  = .otf,
    BoldFont   = XITSMath-Bold,
    BoldItalicFont = XITSMath-Bold
]{XITSMath}
% ⚠️ CJK 等宽**必须用带中文字形的字体**，不能用 cour。
%   cour（Courier New）只有拉丁字形，被 xeCJK 用作 CJK 等宽字体后，
%   代码块里的中文注释全部找不到字形 → 显示为方块
%   （实测 main.pdf 代码块内中文全是 □，日志 320 条 Missing character）。
%   FZSSJW（方正书宋）是项目内嵌字体，含完整 CJK 字形。
\setCJKmonofont[Path=../fonts/, Extension=.TTF]{FZSSJW}
